> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publicar páginas editoriais de lugares (texto + mosaico de fotos no visualizador + informações + mapa + como chegar), sobre um modelo de conteúdo genérico que nasce no `kizuna-core`.

**Architecture:** O core ganha (1) `server/content`: coleções de Markdown com frontmatter validado e mídia ao lado; (2) `shared/maps`: links de rota, distância e URL de mapa; (3) `useCurrentPosition`: posição exata só sob clique; (4) blocos reutilizáveis em `client/components/content`. O starter só define a coleção `lugares`, guarda o conteúdo e monta a página compondo os blocos, no padrão do detalhe do anúncio (não usa o Screen Engine).

**Tech Stack:** Next.js 16 (App Router, Server Components), React 19, TypeScript, Tailwind v4, `react-markdown` (já instalado), lucide-react, Vitest (via `npx`, jsdom nos testes de componente).

**Spec:** `docs/superpowers/specs/2026-10-11-lugares-design.md`

## Global Constraints

- **Zero dependência nova.** O core tem `"dependencies": {}`. Nada de `gray-matter`, Leaflet ou similar.
- **Core é submódulo** (`kizuna-core/`, branch `develop`). Arquivos do core ficam em `kizuna-core/…`; o starter importa por `@kizuna/core/*` (alias para `kizuna-core/src/*`).
- **Não commitar.** Regra do dono do projeto: commit só quando ele pedir. O último passo de cada tarefa é "Checkpoint" (rodar testes e relatar), nunca `git commit`.
- Testes ficam **ao lado do código** (`*.test.ts` / `*.test.tsx`). Componente e hook usam `// @vitest-environment jsdom` na primeira linha.
- Rodar testes do core: `cd kizuna-core && npx vitest run <arquivo>`. Rodar testes do starter: `npx vitest run <arquivo>` na raiz. O Vitest não está nas dependências; o `npx` baixa na primeira vez.
- Documentação do core em **pt-BR**, nomes de arquivo `kebab-case` sem acento, `# Título` único e frontmatter `description`; toda página nova entra no `docs/SUMMARY.md`.
- **Tailwind varre `.md`.** Em plano, spec e docs só use classes válidas do Tailwind, nunca inventadas. Esta entrega também exclui `content/` e `docs/` da varredura (Tarefa 9).
- **Tokens de tema, sem cor nova:** `text-foreground`, `text-muted-foreground`, `border-border`, `bg-muted`, `text-brand`, `rounded-2xl`, como o `AdPhotoMosaic`.
- Coordenadas e posição do usuário **nunca** vão para `localStorage`, servidor ou analytics.
- Campo de URL só aceita `http` e `https` (o valor vira `href`).
- Conteúdo real (endereço, horário, telefone, coordenadas) só entra com `status: aprovado` depois de apurado. Nada inventado; lacuna fica `rascunho`.

```ts
/**
 * Parser de frontmatter — subconjunto pequeno de YAML, sem dependência.
 *
 * Aceita: `chave: valor` (texto simples ou entre aspas), listas com hífen, comentários com `#`.
 * NÃO aceita: objetos aninhados, multilinha (`|`, `>`), coleções inline (`[a]`, `{a: 1}`),
 * âncoras e aliases. Sintaxe fora do subconjunto lança `FrontmatterError` com a linha, em vez de
 * falhar em silêncio. Todo valor sai como string (ou lista de strings); quem tipa é a coleção.
 */

export type FrontmatterValue = string | string[];
export type ParsedDocument = { data: Record<string, FrontmatterValue>; body: string };

export class FrontmatterError extends Error {
  readonly line: number;
  constructor(message: string, line: number) {
    super(`linha ${line}: ${message}`);
    this.name = 'FrontmatterError';
    this.line = line;
  }
}

const KEY_LINE = /^([A-Za-z_][A-Za-z0-9_]*):(?:[ \t]+(.*))?$/;
const LIST_ITEM = /^[ \t]*-[ \t]+(.*)$/;
/** Primeiro caractere que, em YAML, abre sintaxe que este parser não implementa. */
const UNSUPPORTED_START = '|>&*[{';

function afterClosingQuote(rest: string, line: number): void {
  const tail = rest.trim();
  if (tail !== '' && !tail.startsWith('#')) {
    throw new FrontmatterError('texto depois das aspas de fechamento', line);
  }
}

function parseDoubleQuoted(text: string, line: number): string {
  let out = '';
  for (let i = 1; i < text.length; i++) {
    const ch = text[i];
    if (ch === '\\') {
      const next = text[i + 1];
      if (next === undefined) break;
      out += next === 'n' ? '\n' : next === 't' ? '\t' : next;
      i++;
      continue;
    }
    if (ch === '"') {
      afterClosingQuote(text.slice(i + 1), line);
      return out;
    }
    out += ch;
  }
  throw new FrontmatterError('aspas duplas sem fechamento', line);
}

function parseSingleQuoted(text: string, line: number): string {
  let out = '';
  for (let i = 1; i < text.length; i++) {
    const ch = text[i];
    if (ch === "'") {
      if (text[i + 1] === "'") {
        out += "'";
        i++;
        continue;
      }
      afterClosingQuote(text.slice(i + 1), line);
      return out;
    }
    out += ch;
  }
  throw new FrontmatterError('aspas simples sem fechamento', line);
}

function parseScalar(raw: string, line: number): string {
  const text = raw.trim();
  if (text === '') return '';
  const first = text[0];
  if (first === '"') return parseDoubleQuoted(text, line);
  if (first === "'") return parseSingleQuoted(text, line);
  if (UNSUPPORTED_START.includes(first)) {
    throw new FrontmatterError(
      `sintaxe não suportada ("${first}"): use texto simples ou coloque o valor entre aspas`,
      line
    );
  }
  const comment = text.search(/[ \t]#/);
  return (comment === -1 ? text : text.slice(0, comment)).trim();
}

export function parseFrontmatter(source: string): ParsedDocument {
  const text = source.replace(/^﻿/, '').replace(/\r\n?/g, '\n');
  const lines = text.split('\n');
  if (lines[0].trimEnd() !== '---') return { data: {}, body: text };

  let close = -1;
  for (let i = 1; i < lines.length; i++) {
    if (lines[i].trimEnd() === '---') {
      close = i;
      break;
    }
  }
  if (close === -1) {
    throw new FrontmatterError('frontmatter sem a linha "---" de fechamento', 1);
  }

  // Sem protótipo: uma chave `__proto__` ou `constructor` vira dado, nunca toca o Object.
  const data = Object.create(null) as Record<string, FrontmatterValue>;
  let listKey: string | null = null;

  for (let i = 1; i < close; i++) {
    const lineNo = i + 1;
    const line = lines[i];
    if (line.trim() === '' || line.trimStart().startsWith('#')) continue;

    const item = LIST_ITEM.exec(line);
    if (item) {
      if (listKey === null) {
        throw new FrontmatterError('item de lista sem uma chave vazia logo antes', lineNo);
      }
      const current = data[listKey];
      const list = Array.isArray(current) ? current : [];
      list.push(parseScalar(item[1], lineNo));
      data[listKey] = list;
      continue;
    }

    if (/^[ \t]/.test(line)) {
      throw new FrontmatterError('recuo não suportado (objetos aninhados não são aceitos)', lineNo);
    }

    const kv = KEY_LINE.exec(line);
    if (!kv) {
      throw new FrontmatterError(`não entendi "${line.trim()}": esperava "chave: valor"`, lineNo);
    }
    const key = kv[1];
    const rawValue = kv[2] ?? '';
    if (Object.prototype.hasOwnProperty.call(data, key)) {
      throw new FrontmatterError(`chave "${key}" repetida`, lineNo);
    }
    data[key] = parseScalar(rawValue, lineNo);
    listKey = rawValue.trim() === '' ? key : null;
  }

  const body = lines.slice(close + 1).join('\n').replace(/^\n+/, '');
  return { data, body };
}
```

```ts
  | { type: 'number'; required?: boolean; min?: number; max?: number }
  | { type: 'date'; required?: boolean }
  | { type: 'enum'; values: readonly string[]; required?: boolean }
  | { type: 'list'; required?: boolean };

export type FieldValue = string | number | string[];
export type EntryData = Record<string, FieldValue | undefined>;

export type CollectionDef = {
```

```ts
import fs from 'node:fs';
import path from 'node:path';
import { parseFrontmatter, type FrontmatterValue } from './frontmatter';
import type {
  CollectionDef,
  Entry,
  EntryData,
  EntryError,
  FieldDef,
  LoadOptions,
  LoadResult,
} from './types';

const SLUG = /^[a-z0-9]+(-[a-z0-9]+)*$/;
const IMAGE_EXTENSIONS = new Set(['.jpg', '.jpeg', '.png', '.webp']);
const DATE = /^(\d{4})-(\d{2})-(\d{2})$/;

const has = (obj: object, key: string) => Object.prototype.hasOwnProperty.call(obj, key);

export function isValidSlug(slug: string): boolean {
  return SLUG.test(slug);
}

/** Valida a definição em tempo de carga: erro de definição é erro do desenvolvedor, não do conteúdo. */
export function defineCollection(def: CollectionDef): CollectionDef {
  const gate = has(def.fields, def.publishWhen.field) ? def.fields[def.publishWhen.field] : undefined;
  if (!gate || gate.type !== 'enum') {
    throw new Error(
      `coleção "${def.name}": publishWhen.field "${def.publishWhen.field}" precisa ser um campo enum`
    );
  }
  if (!gate.values.includes(def.publishWhen.equals)) {
    throw new Error(
      `coleção "${def.name}": publishWhen.equals "${def.publishWhen.equals}" não está nos values do campo`
    );
  }
  return def;
}

function isRealDate(text: string): boolean {
  const m = DATE.exec(text);
  if (!m) return false;
  const [y, mo, d] = [Number(m[1]), Number(m[2]), Number(m[3])];
  const date = new Date(Date.UTC(y, mo - 1, d));
  return date.getUTCFullYear() === y && date.getUTCMonth() === mo - 1 && date.getUTCDate() === d;
}

function isHttpUrl(text: string): boolean {
  try {
    const u = new URL(text);
    return u.protocol === 'http:' || u.protocol === 'https:';
  } catch {
    return false;
  }
}

function checkField(key: string, def: FieldDef, value: FrontmatterValue, errors: string[]): EntryData[string] {
  const isList = Array.isArray(value);
  switch (def.type) {
    case 'string': {
      if (isList) return void errors.push(`"${key}" deve ser um texto, não uma lista`);
      if (def.max !== undefined && value.length > def.max) {
        errors.push(`"${key}" tem ${value.length} caracteres; o máximo é ${def.max}`);
        return undefined;
      }
      if (def.url && !isHttpUrl(value)) {
        errors.push(`"${key}" deve ser um endereço http ou https`);
        return undefined;
      }
      return value;
    }
    case 'number': {
      if (isList) return void errors.push(`"${key}" deve ser um número, não uma lista`);
      if (value.includes(',')) {
        errors.push(`"${key}" usa vírgula decimal: escreva com ponto (ex.: -15.601)`);
        return undefined;
      }
      const n = Number(value);
      if (!Number.isFinite(n)) return void errors.push(`"${key}" não é um número (recebi "${value}")`);
      if (def.min !== undefined && n < def.min) {
        errors.push(`"${key}" precisa ser no mínimo ${def.min}`);
        return undefined;
      }
      if (def.max !== undefined && n > def.max) {
        errors.push(`"${key}" precisa ser no máximo ${def.max}`);
        return undefined;
      }
      return n;
    }
    case 'date': {
      if (isList || !isRealDate(value)) return void errors.push(`"${key}" deve ser uma data AAAA-MM-DD`);
      return value;
    }
    case 'enum': {
      if (isList || !def.values.includes(value)) {
        errors.push(`"${key}" deve ser um de: ${def.values.join(', ')} (recebi "${isList ? '[lista]' : value}")`);
        return undefined;
      }
      return value;
    }
    case 'list': {
      if (!isList) return void errors.push(`"${key}" deve ser uma lista (um item por linha, com hífen)`);
      return value;
    }
  }
}

function validateData(
  raw: Record<string, FrontmatterValue>,
  fields: Record<string, FieldDef>
): { data: EntryData; errors: string[] } {
  const errors: string[] = [];
  const data: EntryData = {};

  for (const key of Object.keys(raw)) {
    if (!has(fields, key)) errors.push(`campo desconhecido "${key}"`);
  }

  for (const [key, def] of Object.entries(fields)) {
    const value = raw[key];
    const empty = value === undefined || value === '' || (Array.isArray(value) && value.length === 0);
    if (empty) {
      if (def.required) errors.push(`campo obrigatório ausente: "${key}"`);
      continue;
    }
    const checked = checkField(key, def, value, errors);
    if (checked !== undefined) data[key] = checked;
  }

  return { data, errors };
}

function listMedia(def: CollectionDef, slug: string, root: string): string[] {
  let names: string[];
  try {
    names = fs.readdirSync(path.join(root, def.mediaDir, slug));
  } catch {
    return [];
  }
  const prefix = def.mediaUrl.replace(/\/$/, '');
  return names
    .filter((n) => IMAGE_EXTENSIONS.has(path.extname(n).toLowerCase()))
    .sort((a, b) => a.localeCompare(b, 'en', { numeric: true }))
    .map((n) => `${prefix}/${slug}/${encodeURIComponent(n)}`);
}

type ReadResult = { entry: Entry } | { error: EntryError };

function readEntry(def: CollectionDef, slug: string, root: string): ReadResult {
  const file = path.join(root, def.contentDir, slug, 'index.md');
  let source: string;
  try {
    source = fs.readFileSync(file, 'utf8');
  } catch {
    return { error: { slug, file, message: 'index.md não encontrado' } };
  }

  let parsed;
  try {
    parsed = parseFrontmatter(source);
  } catch (e) {
    return { error: { slug, file, message: (e as Error).message } };
  }

  const { data, errors } = validateData(parsed.data, def.fields);
  if (errors.length === 0 && def.validate) {
    const message = def.validate(data);
    if (message) errors.push(message);
  }
  if (errors.length > 0) return { error: { slug, file, message: errors.join('; ') } };

  return { entry: { slug, data, body: parsed.body, media: listMedia(def, slug, root) } };
}

const isPublishable = (def: CollectionDef, entry: Entry) =>
  entry.data[def.publishWhen.field] === def.publishWhen.equals;

/** Lê a coleção inteira. `entries` só traz as publicáveis; `errors` traz as inválidas (com arquivo e motivo). */
export function loadEntries(def: CollectionDef, options: LoadOptions = {}): LoadResult {
  const root = options.root ?? process.cwd();
  const base = path.join(root, def.contentDir);

  let dirs: fs.Dirent[];
  try {
    dirs = fs.readdirSync(base, { withFileTypes: true });
  } catch {
    return { entries: [], errors: [] };
  }

  const entries: Entry[] = [];
  const errors: EntryError[] = [];

  for (const dirent of dirs) {
    if (!dirent.isDirectory()) continue;
    const slug = dirent.name;
    if (slug.startsWith('_') || slug.startsWith('.')) continue;
    if (!isValidSlug(slug)) {
      errors.push({
        slug,
        file: path.join(base, slug),
        message: 'nome de pasta inválido: use a-z, 0-9 e hífen',
      });
      continue;
    }
    const result = readEntry(def, slug, root);
    if ('error' in result) errors.push(result.error);
    else if (isPublishable(def, result.entry)) entries.push(result.entry);
  }

  entries.sort((a, b) => a.slug.localeCompare(b.slug));
  return { entries, errors };
}

/** Uma entrada publicada, ou `null` (inexistente, inválida, não publicada ou slug fora do formato). */
export function getEntry(def: CollectionDef, slug: string, options: LoadOptions = {}): Entry | null {
  if (!isValidSlug(slug)) return null;
  const result = readEntry(def, slug, options.root ?? process.cwd());
  if ('error' in result) {
    if (result.error.message !== 'index.md não encontrado') {
      console.warn(`[content] ${def.name}/${slug}: ${result.error.message}`);
    }
    return null;
  }
  return isPublishable(def, result.entry) ? result.entry : null;
}
```

```ts
export type LatLng = { lat: number; lng: number };

export type DirectionsInput = {
  destination?: LatLng | null;
  address?: string | null;
  /** Posição do usuário. Só o Google Maps aceita origem por link; o Waze usa o GPS do aparelho. */
  origin?: LatLng | null;
};

export type DirectionsLinks = { google: string | null; waze: string | null };

const pair = (p: LatLng) => `${p.lat},${p.lng}`;

/** Links oficiais de rota (sem chave de API). Coordenadas têm preferência; sem elas usa o endereço. */
export function directionsLinks({ destination, address, origin }: DirectionsInput): DirectionsLinks {
  const text = address?.trim() ?? '';
  const target = destination ? pair(destination) : text;
  if (!target) return { google: null, waze: null };

  const encoded = encodeURIComponent(target);
  const google =
    `https://www.google.com/maps/dir/?api=1&destination=${encoded}` +
    (origin ? `&origin=${encodeURIComponent(pair(origin))}` : '');
  const waze = destination
    ? `https://waze.com/ul?ll=${encoded}&navigate=yes`
    : `https://waze.com/ul?q=${encoded}&navigate=yes`;

  return { google, waze };
}
```

```ts
import type { LatLng } from './directions';

const EARTH_RADIUS_KM = 6371.0088;
const toRad = (deg: number) => (deg * Math.PI) / 180;

/** Distância em linha reta (haversine). NÃO é distância de percurso. */
export function distanceKm(a: LatLng, b: LatLng): number {
  const dLat = toRad(b.lat - a.lat);
  const dLng = toRad(b.lng - a.lng);
  const h =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRad(a.lat)) * Math.cos(toRad(b.lat)) * Math.sin(dLng / 2) ** 2;
  return 2 * EARTH_RADIUS_KM * Math.asin(Math.min(1, Math.sqrt(h)));
}

/** "430 m", "6,3 km", "12 km" (vírgula decimal, pt-BR). */
export function formatDistanceKm(km: number): string {
  if (km < 1) return `${Math.round(km * 1000)} m`;
  if (km < 10) return `${km.toFixed(1).replace('.', ',')} km`;
  return `${Math.round(km)} km`;
}
```

```ts
/**
 * URL da incorporação do OpenStreetMap (sem chave): caixa de `span` graus ao redor do ponto, com
 * marcador. `span` padrão 0,006 grau (cerca de 650 m).
 */
export function osmEmbedUrl({ lat, lng, span = 0.006 }: { lat: number; lng: number; span?: number }): string {
  const half = span / 2;
  const bbox = [lng - half, lat - half, lng + half, lat + half].map(round).join(',');
  const params = [
    `bbox=${encodeURIComponent(bbox)}`,
    'layer=mapnik',
    `marker=${encodeURIComponent(`${lat},${lng}`)}`,
  ];
  return `https://www.openstreetmap.org/export/embed.html?${params.join('&')}`;
}
```

```ts
'use client';

import { useCallback, useEffect, useRef, useState } from 'react';

export type PositionStatus = 'idle' | 'asking' | 'granted' | 'denied' | 'unavailable';
export type CurrentPosition = { lat: number; lng: number; accuracy: number | null };

const GEOLOCATION_PERMISSION_DENIED = 1;

/**
 * Posição EXATA do aparelho, pedida só quando `request()` é chamado (em resposta a um clique —
 * nunca ao montar). Diferente de `useUserLocation`, que resolve cidade/UF e persiste para filtrar
 * o site: aqui as coordenadas ficam só em memória, nunca vão para `localStorage`, servidor ou
 * analytics. Sem a API, negado ou com falha: o consumidor segue sem posição.
 */
export function useCurrentPosition() {
  const [status, setStatus] = useState<PositionStatus>('idle');
  const [position, setPosition] = useState<CurrentPosition | null>(null);
  const mounted = useRef(true);

  useEffect(() => {
    mounted.current = true;
    return () => {
      mounted.current = false;
    };
  }, []);

  const request = useCallback(() => {
    if (typeof navigator === 'undefined' || !navigator.geolocation) {
      setStatus('unavailable');
      return;
    }
    setStatus('asking');
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        if (!mounted.current) return;
        setPosition({
          lat: pos.coords.latitude,
          lng: pos.coords.longitude,
          accuracy: pos.coords.accuracy ?? null,
        });
        setStatus('granted');
      },
      (err) => {
        if (!mounted.current) return;
        setPosition(null);
        setStatus(err.code === GEOLOCATION_PERMISSION_DENIED ? 'denied' : 'unavailable');
      },
      { enableHighAccuracy: false, timeout: 10_000, maximumAge: 60_000 }
    );
  }, []);

  return { status, position, request };
}
```

```tsx
'use client';

import { useState } from 'react';
import { Check, Copy, LocateFixed, Loader2, Navigation } from 'lucide-react';
import { cn } from '../../../lib/utils';
import { directionsLinks, distanceKm, formatDistanceKm, type LatLng } from '../../../shared/maps';
import { useCurrentPosition } from '../../hooks/use-current-position';

export type DirectionsCardProps = {
  destination?: LatLng | null;
  address?: string | null;
  tips?: string | null;
  className?: string;
};

const BUTTON =
  'inline-flex items-center justify-center gap-2 rounded-full border border-border bg-background px-4 py-2 text-sm font-medium text-foreground transition hover:bg-muted';

/**
 * "Como chegar": links de rota (Google Maps, Waze), copiar endereço e, SÓ sob clique, distância em
 * linha reta a partir da posição do usuário. A posição fica em memória (ver `useCurrentPosition`)
 * e só sai do navegador quando o usuário abre o Google Maps, com a origem no link.
 */
export function DirectionsCard({ destination = null, address = null, tips = null, className }: DirectionsCardProps) {
  const { status, position, request } = useCurrentPosition();
  const [copied, setCopied] = useState(false);

  const origin = status === 'granted' && position ? { lat: position.lat, lng: position.lng } : null;
  const links = directionsLinks({ destination, address, origin });
  const hasRoute = links.google !== null || links.waze !== null;
  if (!hasRoute && !tips) return null;

  const distance = origin && destination ? formatDistanceKm(distanceKm(origin, destination)) : null;
  const canMeasure = destination !== null && status !== 'granted';

  async function copyAddress() {
    if (!address) return;
    try {
      await navigator.clipboard.writeText(address);
      setCopied(true);
      window.setTimeout(() => setCopied(false), 2000);
    } catch {
      // Sem permissão de área de transferência: o botão só não confirma.
    }
  }

  return (
    <section className={cn('rounded-2xl border border-border bg-muted/40 p-5', className)}>
      <h2 className="text-base font-bold text-foreground">Como chegar</h2>

      {address ? (
        <div className="mt-3 flex items-start justify-between gap-3">
          <p className="text-sm leading-6 text-foreground">{address}</p>
          <button type="button" onClick={copyAddress} className={cn(BUTTON, 'shrink-0 px-3')}>
            {copied ? <Check className="h-4 w-4" aria-hidden="true" /> : <Copy className="h-4 w-4" aria-hidden="true" />}
            {copied ? 'Copiado' : 'Copiar'}
          </button>
        </div>
      ) : null}

      {hasRoute ? (
        <div className="mt-4 flex flex-wrap gap-2">
          {links.google ? (
            <a href={links.google} target="_blank" rel="noopener noreferrer" className={BUTTON}>
              <Navigation className="h-4 w-4 text-brand" aria-hidden="true" />
              Abrir no Google Maps
            </a>
          ) : null}
          {links.waze ? (
            <a href={links.waze} target="_blank" rel="noopener noreferrer" className={BUTTON}>
              <Navigation className="h-4 w-4 text-brand" aria-hidden="true" />
              Abrir no Waze
            </a>
          ) : null}
        </div>
      ) : null}

      {distance ? (
        <p className="mt-4 text-sm text-foreground">
          Cerca de <strong>{distance}</strong> de você, em linha reta.
        </p>
      ) : null}

      {canMeasure && status !== 'asking' ? (
        <div className="mt-4">
          <button type="button" onClick={request} className={BUTTON}>
            <LocateFixed className="h-4 w-4 text-brand" aria-hidden="true" />
            Ver distância e rota a partir de você
          </button>
          <p className="mt-2 text-xs leading-5 text-muted-foreground">
            Usamos sua localização só neste aparelho, para calcular a distância. Não guardamos.
          </p>
        </div>
      ) : null}

      {status === 'asking' ? (
        <p className="mt-4 inline-flex items-center gap-2 text-sm text-muted-foreground">
          <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
          Buscando sua localização…
        </p>
      ) : null}

      {status === 'denied' ? (
        <p className="mt-3 text-xs leading-5 text-muted-foreground">
          Sem acesso à sua localização. A rota continua disponível pelos botões acima.
        </p>
      ) : null}
      {status === 'unavailable' ? (
        <p className="mt-3 text-xs leading-5 text-muted-foreground">
          Não foi possível obter sua localização agora.
        </p>
      ) : null}

      {tips ? <p className="mt-4 text-sm leading-6 text-foreground">{tips}</p> : null}
    </section>
  );
}
```

````markdown
# Modelo de conteúdo

Uma **coleção** é uma pasta de artigos em Markdown, um por subpasta, com frontmatter validado e
fotos ao lado. O core lê, valida e lista; o projeto decide o que a coleção é (lugares, guias,
roteiros) e como a página fica. Sem banco e sem painel: o conteúdo vive no git.

## Estrutura de pastas

```
content/lugares/lua-morena/index.md     texto + frontmatter
public/lugares/lua-morena/01.jpg ...    fotos; a ordem é o nome do arquivo (ordem natural)
```

- O nome da pasta é o **slug**: só `a-z`, `0-9` e hífen.
- Pasta que começa com `_` ou `.` é ignorada (use para modelos).
- Fotos ficam em `public/` porque o Next só serve arquivos de lá. Aceita `.jpg`, `.jpeg`, `.png`
  e `.webp`, em qualquer caixa.

## Definir a coleção

```ts
import { defineCollection } from '@kizuna/core/server/content';

export const LUGARES = defineCollection({
  name: 'lugares',
  contentDir: 'content/lugares',
  mediaDir: 'public/lugares',
  mediaUrl: '/lugares',
  fields: {
    titulo: { type: 'string', required: true },
    resumo: { type: 'string', required: true, max: 155 },
    status: { type: 'enum', values: ['rascunho', 'revisao', 'aprovado'], required: true },
    atualizado_em: { type: 'date', required: true },
    latitude: { type: 'number', min: -90, max: 90 },
    longitude: { type: 'number', min: -180, max: 180 },
    site: { type: 'string', url: true },
    horario: { type: 'list' },
  },
  publishWhen: { field: 'status', equals: 'aprovado' },
  // Regra entre campos, opcional: devolve a mensagem de erro ou null.
  validate: (d) =>
    (d.latitude === undefined) !== (d.longitude === undefined)
      ? 'latitude e longitude devem vir juntas'
      : null,
});
```

| Tipo | Opções | Observação |
|---|---|---|
| `string` | `required`, `max`, `url` | `url` aceita só `http` e `https` (o valor costuma virar `href`) |
| `number` | `required`, `min`, `max` | escreva com **ponto** decimal; vírgula é erro |
| `date` | `required` | formato `AAAA-MM-DD`, data real |
| `enum` | `values`, `required` | |
| `list` | `required` | um item por linha, com hífen |

`publishWhen` aponta para um campo `enum`: só a entrada com aquele valor aparece.

## Ler

```ts
import { loadEntries, getEntry } from '@kizuna/core/server/content';

const { entries, errors } = loadEntries(LUGARES); // publicadas + inválidas (com arquivo e motivo)

```

`Entry` traz `slug`, `data` (tipada por campo), `body` (Markdown sem frontmatter) e `media`

devolve `null` e registra um aviso no log do servidor. Um slug fora do formato devolve `null` sem
tocar o disco.

## Frontmatter aceito

Subconjunto pequeno de YAML: `chave: valor` (com ou sem aspas), listas com hífen e comentários
com `#`.

```yaml
titulo: "Lua Morena"
status: rascunho   # comentário
horario:
  - "Qui a sáb, 22h às 5h"
```

Não são aceitos objetos aninhados, multilinha (`|`, `>`), coleções inline (`[a, b]`) nem
âncoras. O erro aponta a linha. Valor que começa com `[` ou `{` precisa de aspas.
````

```ts
import { defineCollection, type Entry, type EntryData } from '@kizuna/core/server/content';

/**
 * Coleção "lugares": artigos editoriais de lugares (casas noturnas, casas de festas, shoppings).
 * O mecanismo (leitura, validação, fotos) é do core; aqui só se diz o que um lugar é.
 * Conteúdo em `content/lugares/<slug>/index.md`, fotos em `public/lugares/<slug>/`.
 */
export const LUGARES = defineCollection({
  name: 'lugares',
  contentDir: 'content/lugares',
  mediaDir: 'public/lugares',
  mediaUrl: '/lugares',
  fields: {
    titulo: { type: 'string', required: true },
    categoria: { type: 'string', required: true },
    cidade: { type: 'string', required: true },
    resumo: { type: 'string', required: true, max: 155 },
    status: { type: 'enum', values: ['rascunho', 'revisao', 'aprovado'], required: true },
    atualizado_em: { type: 'date', required: true },
    endereco: { type: 'string' },
    latitude: { type: 'number', min: -90, max: 90 },
    longitude: { type: 'number', min: -180, max: 180 },
    horario: { type: 'list' },
    telefone: { type: 'string' },
    instagram: { type: 'string' },
    site: { type: 'string', url: true },
    preco: { type: 'string' },
    dicas_chegada: { type: 'string' },
    creditos: { type: 'string' },
  },
  publishWhen: { field: 'status', equals: 'aprovado' },
  validate: (d) =>
    (d.latitude === undefined) !== (d.longitude === undefined)
      ? 'latitude e longitude devem vir juntas'
      : null,
});

export type LugarFactKey = 'endereco' | 'horario' | 'telefone' | 'site' | 'instagram' | 'preco';
export type LugarActionKind = 'rota' | 'ligar' | 'site';

export type LugarView = {
  title: string;
  kicker: string;
  summary: string;
  updatedAt: string;
  body: string;
  photos: string[];
  destination: { lat: number; lng: number } | null;
  address: string | null;
  tips: string | null;
  credits: string | null;
  facts: { key: LugarFactKey; label: string; value: string | string[]; href?: string }[];
  actions: { kind: LugarActionKind; label: string; href: string; external?: boolean }[];
};

const text = (data: EntryData, key: string): string | null => {
  const v = data[key];
  return typeof v === 'string' && v.trim() !== '' ? v.trim() : null;
};
const num = (data: EntryData, key: string): number | null => {
  const v = data[key];
  return typeof v === 'number' ? v : null;
};
const list = (data: EntryData, key: string): string[] => {
  const v = data[key];
  return Array.isArray(v) ? v.filter((s) => s.trim() !== '') : [];
};

/** `tel:` só com 8 ou mais dígitos; senão o telefone aparece como texto, sem link. */
function telHref(raw: string): string | undefined {
  const digits = raw.replace(/\D/g, '');
  if (digits.length < 8) return undefined;
  return `tel:${raw.trim().startsWith('+') ? '+' : ''}${digits}`;
}

/** Aceita `@perfil`, `perfil` ou a URL do perfil; devolve o usuário ou `null` se não for válido. */
function instagramHandle(raw: string): string | null {
  const handle = raw
    .trim()
    .replace(/^https?:\/\/(www\.)?instagram\.com\//i, '')
    .replace(/^@/, '')
    .replace(/\/+$/, '');
  return /^[A-Za-z0-9._]{1,30}$/.test(handle) ? handle : null;
}

function siteLabel(url: string): string {
  try {
    return new URL(url).hostname.replace(/^www\./, '');
  } catch {
    return url;
  }
}

/** Reduz a entrada a dados prontos para a página (sem JSX): fácil de testar e de compor. */
export function buildLugarView(entry: Entry, cityName: string): LugarView {
  const d = entry.data;
  const lat = num(d, 'latitude');
  const lng = num(d, 'longitude');
  const destination = lat !== null && lng !== null ? { lat, lng } : null;
  const address = text(d, 'endereco');
  const phone = text(d, 'telefone');
  const site = text(d, 'site');
  const insta = text(d, 'instagram');
  const handle = insta ? instagramHandle(insta) : null;
  const phoneHref = phone ? telHref(phone) : undefined;
  const hours = list(d, 'horario');
  const price = text(d, 'preco');

  const facts: LugarView['facts'] = [];
  if (address) facts.push({ key: 'endereco', label: 'Endereço', value: address });
  if (hours.length > 0) facts.push({ key: 'horario', label: 'Horário', value: hours });
  if (phone) facts.push({ key: 'telefone', label: 'Telefone', value: phone, href: phoneHref });
  if (site) facts.push({ key: 'site', label: 'Site', value: siteLabel(site), href: site });
  if (insta) {
    facts.push(
      handle
        ? { key: 'instagram', label: 'Instagram', value: `@${handle}`, href: `https://instagram.com/${handle}` }
        : { key: 'instagram', label: 'Instagram', value: insta }
    );
  }
  if (price) facts.push({ key: 'preco', label: 'Preço', value: price });

  const actions: LugarView['actions'] = [];
  if (destination || address) actions.push({ kind: 'rota', label: 'Como chegar', href: '#como-chegar' });
  if (phoneHref) actions.push({ kind: 'ligar', label: 'Ligar', href: phoneHref });
  if (site) actions.push({ kind: 'site', label: 'Site', href: site, external: true });

  return {
    title: text(d, 'titulo') ?? entry.slug,
    kicker: `${text(d, 'categoria') ?? ''} · ${cityName}`,
    summary: text(d, 'resumo') ?? '',
    updatedAt: text(d, 'atualizado_em') ?? '',
    body: entry.body,
    photos: entry.media,
    destination,
    address,
    tips: text(d, 'dicas_chegada'),
    credits: text(d, 'creditos'),
    facts,
    actions,
  };
}
```

```tsx
/** Lugar publicado E da cidade da URL; qualquer outro caso vira 404. */
async function load(cidade: string, slug: string) {
  const city = await resolveCitySlug(cidade);
  if (!city) return null;
  const entry = getEntry(LUGARES, slug);
  if (!entry || entry.data.cidade !== cidade) return null;
  return { city, entry };
}

export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { cidade, slug } = await params;
  const found = await load(cidade, slug);
  if (!found) return { title: 'Lugar não encontrado' };
  const view = buildLugarView(found.entry, found.city.name);
  return buildMetadata(SEO_SITE, {
    title: `${view.title} em ${found.city.name}`,
    description: view.summary,
    path: cityPath(found.city, `/lugares/${slug}`),
    image: view.photos[0] ?? null,
  });
}

export default async function LugarRoute({ params }: Props) {
  const { cidade, slug } = await params;
  const found = await load(cidade, slug);
  if (!found) notFound();
  return <LugarPage view={buildLugarView(found.entry, found.city.name)} />;
}
```
